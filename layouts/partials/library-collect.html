{{- /* The whole Library catalog: every published entry stored under content/library/ (inclusion is implicit there) plus any page elsewhere on the site with `library.include: true`. Deduplicated by canonical page. Sorted by weight (when supplied) → sort_title → title. Call: {{ $entries := partial "library-collect.html" (dict "ctx" .) }} */ -}} {{- $ctx := .ctx -}} {{- $inSection := where $ctx.Site.RegularPages "Section" "library" -}} {{- $included := where $ctx.Site.RegularPages "Params.library.include" true -}} {{- $all := $inSection | union $included -}} {{- /* stable ordering: 6-digit weight (default high) then sort_title|title */ -}} {{- $keyed := slice -}} {{- range $all -}} {{- $w := int (.Params.weight | default 100000) -}} {{- $k := printf "%06d|%s" $w (lower (.Params.sort_title | default .Title)) -}} {{- $keyed = $keyed | append (dict "p" . "k" $k) -}} {{- end -}} {{- $out := slice -}} {{- range (sort $keyed "k") }}{{ $out = $out | append .p }}{{ end -}} {{- return $out -}}
