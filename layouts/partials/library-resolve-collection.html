{{- /* Resolve which registered Collection (data/library/collections.yaml) a Library page belongs to — either a Collection's own section index page or one of its Entries — by longest-prefix match of the page's content-relative File.Path against each Collection's source.section (adapter: hugo only). One resolution mechanism used everywhere a Library template needs "which Collection is this" — see docs/library-v2.md § 2.2/§ 3. Call: {{ $coll := partial "library-resolve-collection.html" . }} Returns the matched Collection dict (data/library/collections.yaml shape), or fails the build if no Collection's source.section prefixes this page's path. */ -}} {{- $page := . -}} {{- $registry := hugo.Data.library.collections -}} {{- $coll := dict -}} {{- $bestLen := 0 -}} {{- range $registry -}} {{- if and (eq .source.adapter "hugo") .source.section (hasPrefix $page.File.Path (printf "%s/" .source.section)) (gt (len .source.section) $bestLen) -}} {{- $coll = . }}{{ $bestLen = len .source.section -}} {{- end -}} {{- end -}} {{- if not $coll.id }}{{ errorf "library-resolve-collection.html: no Collection in data/library/collections.yaml matches %q" $page.File.Path }}{{ end -}} {{- return $coll -}}
