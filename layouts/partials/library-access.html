{{- /* Render an entry's `access` array as labelled links, capped to a single row: catalog rows never wrap to multiple lines. Context is the page. Each entry: {label, url|file, kind}. `hosted-file` resolves a bundle file to a download link; other (external) kinds link out with ↗. The kind's external-ness comes from data/library.yaml. The author-supplied label is the visible text. Beyond `$max` items, a single "and N more" link to the entry page replaces the rest (the single-entry page itself always shows every access item in full — this cap is catalog-list-only). `$max` is kept at 1 deliberately: with long labels and several access items (e.g. Rainforest's nine), a higher cap risks the trailing "and N more" link — the one telling the reader more exists — being clipped by the row's `overflow: hidden`/`nowrap` (see library.css); one link plus the count reliably fits. Renders nothing without `access`. Call: {{ partial "library-access.html" . }} */ -}} {{- $page := . -}} {{- $max := 1 -}} {{- $ext := dict -}}{{ $verbs := dict -}} {{- range hugo.Data.library.access_kinds }}{{ $ext = merge $ext (dict .kind .external) }}{{ $verbs = merge $verbs (dict .kind .verb) }}{{ end -}} {{- with $page.Params.access -}} {{- $items := . -}} {{- $total := len $items -}} {{- $shown := $items -}}{{ if gt $total $max }}{{ $shown = first $max $items }}{{ end -}}
    {{- range $shown }}{{ $a := . }} {{- $kind := $a.kind | default "other" -}} {{- $label := $a.label | default (index $verbs $kind) | default "Link" -}} {{- if eq $kind "hosted-file" }} {{- with $a.file }}{{ with ($page.Resources.GetMatch .) }}
  • {{ $label }}
    • {{- end }}{{ end }} {{- else }}{{ with $a.url }} {{- $isExt := index $ext $kind }}
  • {{ $label }}{{ if $isExt }} ↗{{ end }}
    • {{- end }}{{ end }} {{- end }} {{- if gt $total $max }}
  • and {{ sub $total $max }} more
    • {{- end }}
{{- end -}}
